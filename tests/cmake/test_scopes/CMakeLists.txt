cmake_minimum_required(VERSION 3.25)
project(DraxulTestScopes NONE)
enable_testing()
include(${DRAXUL_SOURCE_ROOT}/cmake/DraxulTestScopes.cmake)
option(ENABLE_PRODUCT "Mount the fixture product" ON)
foreach(owner app-shell host-api server)
    add_custom_target(test-${owner})
    draxul_register_test_target(test-${owner} ${owner})
    add_test(NAME test-${owner} COMMAND ${CMAKE_COMMAND} -E true)
    set_tests_properties(test-${owner} PROPERTIES LABELS "${owner};integration")
endforeach()
if(ENABLE_PRODUCT)
    foreach(owner model parser)
        add_custom_target(test-megacity-${owner})
        draxul_register_test_target(test-megacity-${owner} megacity)
        add_test(NAME test-megacity-${owner} COMMAND ${CMAKE_COMMAND} -E true)
        set_tests_properties(test-megacity-${owner} PROPERTIES LABELS "megacity;unit")
    endforeach()
elseif(TARGET draxul-tests-megacity)
    message(FATAL_ERROR "Disabled product registered a build aggregate")
endif()
# Python/script integrations have no test executable but still belong in scope.
add_test(NAME script-integration COMMAND ${CMAKE_COMMAND} -E true)
set_tests_properties(script-integration PROPERTIES LABELS "integration;python")
draxul_register_test_scopes()
get_target_property(core draxul-tests-core MANUALLY_ADDED_DEPENDENCIES)
if(NOT "${core}" STREQUAL "test-app-shell;test-host-api;test-server")
    message(FATAL_ERROR "Incomplete core build membership: ${core}")
endif()
if(ENABLE_PRODUCT)
    get_target_property(product draxul-tests-megacity MANUALLY_ADDED_DEPENDENCIES)
    if(NOT "${product}" STREQUAL "test-megacity-model;test-megacity-parser")
        message(FATAL_ERROR "Incomplete product build membership: ${product}")
    endif()
endif()
